# 微信机器人项目 bot.js

```
import { WechatyBuilder } from 'wechaty'  
const bot = WechatyBuilder.build() // 创建 Wechaty 实例  
bot  
.on('scan', (qrcode, status) => console.log(`扫码登录中...\n${status}: https://wechaty.js.org/qrcode/${encodeURIComponent(qrcode)}`))  
.on('login', user => console.log(`用户 ${user} 登录成功!`))  
.on('message', message => console.log(`收到消息: ${message}`))  
bot.start()
```
